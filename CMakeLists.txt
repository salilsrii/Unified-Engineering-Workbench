cmake_minimum_required(VERSION 3.16)
project(PassWorkbench VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(workbench src/cpp/main.cpp)
target_link_libraries(workbench PRIVATE Threads::Threads)
if(MSVC)
  set_property(TARGET workbench PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(workbench PRIVATE /W4 /permissive-)
  target_link_libraries(workbench PRIVATE ws2_32 shell32)
else()
  target_compile_options(workbench PRIVATE -Wall -Wextra -Wpedantic)
endif()
